DROP DATABASE IF EXISTS crud_manager;

CREATE DATABASE crud_manager;

USE crud_manager;

CREATE TABLE instrutores (
    id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(150) NOT NULL,
    sexo ENUM('M', 'F') NOT NULL,
    email VARCHAR(150) NOT NULL,
    telefone VARCHAR(20) NOT NULL,
    cnh VARCHAR(20) NOT NULL,
    password VARCHAR(255) NOT NULL
);
CREATE TABLE alunos (
    id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(150) NOT NULL,
    data_nascimento DATE NOT NULL,
    cpf VARCHAR(14) NOT NULL,
    categoria ENUM('ACC', 'A', 'B', 'AB', 'C', 'D', 'E') NOT NULL,
    instrutor_id INT NOT NULL,

    FOREIGN KEY (instrutor_id)
    REFERENCES instrutores(id)
);

CREATE TABLE veiculos (
    id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    modelo VARCHAR(100) NOT NULL,
    placa VARCHAR(10) NOT NULL,
    ano INT NOT NULL,
    categoria ENUM('ACC', 'A', 'B', 'C', 'D', 'E') NOT NULL,
    data_aquisicao DATE NOT NULL,
    situacao ENUM('Disponível', 'Em manutenção', 'Em uso') NOT NULL,
    instrutor_id INT NOT NULL,

    FOREIGN KEY (instrutor_id)
    REFERENCES instrutores(id)
);

CREATE TABLE aulas (
    id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    data_aula DATE NOT NULL,
    hora_inicio TIME NOT NULL,
    duracao_minutos INT NOT NULL,
    status ENUM('Agendada', 'Realizada', 'Cancelada') NOT NULL,
    observacoes TEXT,
    aluno_id INT NOT NULL,
    instrutor_id INT NOT NULL,
    veiculo_id INT NOT NULL,

    FOREIGN KEY (aluno_id) REFERENCES alunos(id),
    FOREIGN KEY (instrutor_id) REFERENCES instrutores(id),
    FOREIGN KEY (veiculo_id) REFERENCES veiculos(id)
);

CREATE TABLE pagamentos (
    id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    valor DECIMAL(10,2) NOT NULL,
    data_pagamento DATE NOT NULL,
    forma_pagamento ENUM('Dinheiro', 'Pix', 'Cartão', 'Transferência') NOT NULL,
    status ENUM('Pendente', 'Pago', 'Cancelado') NOT NULL,
    descricao TEXT,
    numero_parcelas INT NOT NULL DEFAULT 1,
    aluno_id INT NOT NULL,

    FOREIGN KEY (aluno_id) REFERENCES alunos(id)
);
